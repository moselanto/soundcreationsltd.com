# ============================================================================
# Sound Creations Ltd - uploads folder
# File: /public_html/wp-content/uploads/.htaccess   (a NEW, separate file)
# Added: 2026-10-19
#
# The uploads folder only ever holds media and documents. If an attacker ever
# manages to upload a PHP file (through a vulnerable plugin, for example),
# this stops the server from running it. It is the single most effective
# file-level hardening step for WordPress and does not affect images, video,
# PDFs or the enquiry-form attachments.
# ============================================================================

<FilesMatch "\.(php[0-9]?|phtml|phar|pht|phps|cgi|pl|py|sh|asp|aspx|jsp)$">
  Require all denied
</FilesMatch>

# Belt and braces: even if the handler is reached, do not execute scripts here.
<IfModule mod_php.c>
  php_flag engine off
</IfModule>

Options -Indexes -ExecCGI
