# ============================================================================
# Sound Creations Ltd - uploads folder protection
# File: /public_html/wp-content/uploads/.htaccess
# Added: 2026-10-19
#
# The uploads folder should only ever hold media and documents. If a script
# ever gets uploaded here (for example through a vulnerable plugin), this
# stops the server from running it. Images, video, PDFs and enquiry-form
# attachments are not affected.
#
# Uses only "FilesMatch + Require all denied", the same kind of rule the
# root .htaccess already runs on this server.
# ============================================================================

<FilesMatch "\.(php|php3|php4|php5|php7|php8|phtml|phar|pht|phps|cgi|pl|py|sh|asp|aspx|jsp)$">
  Require all denied
</FilesMatch>
